# Multilangue FR/EN — plan d'implémentation

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** rendre SpinForge intégralement bilingue français/anglais, la langue étant déduite du navigateur et surchargeable par le joueur.

**Architecture:** une couche maison `src/i18n/` de deux catalogues TypeScript plats, dont l'anglais est typé sur le français — une clé manquante casse le build. `t()` lit un singleton de module, comme `audio.ts`, ce qui laisse `axisLine()`, `rankLabel()` et `formatCredits()` sans paramètre de locale. Les libellés joueur quittent `src/sim/` ; les catalogues de `src/content/` cessent de porter du texte et voient leurs clés dérivées de leurs identifiants.

**Tech Stack:** TypeScript strict, React 19, Vite 6, Vitest 2 (`environment: 'node'`), `Intl.PluralRules` et `Intl.NumberFormat` — aucune dépendance ajoutée.

**Spec:** `docs/superpowers/specs/2026-08-31-i18n-fr-en-design.md`

## Global Constraints

- `src/sim/` reste pur et déterministe : il n'importe **jamais** `src/i18n/`, dans aucun sens.
- Aucune dépendance npm ajoutée.
- `src/i18n/index.ts` reste **libre de React et de DOM** : il est testé en `environment: 'node'`. `document.documentElement.lang` est posé par `main.tsx` et `App.tsx`, jamais par `setLang`.
- Vitest ne collecte que `src/**/*.test.ts` — **pas les `.tsx`**. Tout test portant sur `tx.tsx` vit dans un `.test.ts` et construit ses nœuds avec `createElement`.
- `tsconfig` a `noUnusedLocals` et `noUnusedParameters` : tout import ou paramètre laissé derrière casse `npm run build`.
- Règle IP inchangée : aucun nom officiel Beyblade, en anglais pas plus qu'en français.
- Clé de persistance : `spinforge.lang`. Le schéma de sauvegarde reste **4**.
- Tout accès à `localStorage` est en `try/catch`, comme `src/storage/localSave.ts`.

- [ ] **Step 5 : écrire `src/i18n/index.ts`**

```ts
import { en } from './en';
import { fr, type MessageKey as Key } from './fr';

export type Lang = 'fr' | 'en';
export type { MessageKey } from './fr';
/** Distribue sur l'union — d'où le paramètre nommé plutôt que `MessageKey`
 *  directement : un type concret ne distribue pas dans un type conditionnel. */
type BaseOf<K> = K extends `${infer B}.one` ? B : never;
export type PluralKey = BaseOf<Key>;

const KEY = 'spinforge.lang';
const CATALOGS: Record<Lang, Record<Key, string>> = { fr, en };

function isLang(v: unknown): v is Lang {
  return v === 'fr' || v === 'en';
}

/** Cœur pur de la détection, pour qu'elle soit testable sans navigateur :
 *  le choix stocké prime, puis les préférences du navigateur, puis l'anglais —
 *  un joueur ni francophone ni anglophone le lira plus probablement. */
export function pickLang(stored: string | null, preferred: readonly string[]): Lang {
  if (isLang(stored)) return stored;
  for (const tag of preferred) {
    const base = tag.toLowerCase().split('-')[0];
    if (isLang(base)) return base;
  }
  return 'en';
}

function read(): string | null {
  try {
    return localStorage.getItem(KEY);
  } catch {
    return null; // Stockage refusé : le navigateur tranchera seul.
  }
}

export function detectLang(): Lang {
  const nav = typeof navigator === 'undefined' ? undefined : navigator;
  return pickLang(read(), nav?.languages ?? (nav?.language ? [nav.language] : []));
}

let current: Lang = detectLang();

export function getLang(): Lang {
  return current;
}

/** Ne touche pas au DOM : `document.documentElement.lang` est posé par `main.tsx`
 *  et par le sélecteur, ce qui garde ce module testable en environnement node. */
export function setLang(lang: Lang): void {
  current = lang;
  try {
    localStorage.setItem(KEY, lang);
  } catch {
    // Idem : la langue vaut alors pour la session seulement.
  }
}

export function raw(key: MessageKey): string {
  return CATALOGS[current][key];
}

/** Le repli sur `.other` n'est pas décoratif : le CLDR récent renvoie `many`
 *  pour le français au-delà du million, et le typage n'exige que `.one`/`.other`. */
export function rawPlural(base: PluralKey, count: number): string {
  const category = new Intl.PluralRules(current).select(count);
  const catalog = CATALOGS[current];
  const key = `${base}.${category}` as MessageKey;
  return catalog[key] ?? catalog[`${base}.other` as MessageKey];
}

function interpolate(template: string, vars: Record<string, string | number>): string {
  return template.replace(/\{(\w+)\}/g, (whole, name: string) =>
    name in vars ? String(vars[name]) : whole,
  );
}

export function t(key: MessageKey, vars?: Record<string, string | number>): string {
  const template = raw(key);
  return vars ? interpolate(template, vars) : template;
}

export function tn(base: PluralKey, count: number, vars?: Record<string, string | number>): string {
  return interpolate(rawPlural(base, count), { n: count, ...vars });
}

/** Le suffixe est une clé : le français veut « 12,4 k » avec espace, l'anglais
 *  « 12.4k » sans. Le séparateur décimal vient d'`Intl`, plus d'un `replace`. */
export function formatCredits(n: number): string {
  const fixed = (value: number, digits: number) =>
    new Intl.NumberFormat(current, {
      minimumFractionDigits: digits, maximumFractionDigits: digits, useGrouping: false,
    }).format(value);
  if (n < 1000) return Math.floor(n).toString();
  if (n < 999950) return fixed(n / 1000, 1) + t('format.thousand');
  return fixed(n / 1e6, 2) + t('format.million');
}
```

import { afterEach, describe, expect, it } from 'vitest';
import { setLang } from './index';
import { tx, txn } from './tx';

afterEach(() => setLang('fr'));

/** Les enfants d'un Fragment construit par `createElement(f, null, ...parts)`. */

  return (el.props as { children: unknown[] }).children;
}

describe('tx', () => {
  it('découpe le gabarit autour des nœuds insérés, dans l’ordre', () => {
    const strong = createElement('strong', null, 'Fonce dans l’adversaire');
    const out = parts(tx('combat.hint.body', { charge: strong }));
    expect(out[0]).toBe('Glisse le doigt n’importe où pour la piloter. ');
    expect(out[1]).toBe(strong);
    expect(out[2]).toBe(' : qui charge casse plus et encaisse moins.');

  // fragment vide parasite au bord.

    const rank = createElement('span', null, 'Épique');
    const out = parts(tx('chest.bestRank', { rank }));
    expect(out.filter((p) => p === '')).toHaveLength(0);
    expect(out).toContain(rank);
  });

  it('laisse le gabarit intact quand aucune variable n’est fournie', () => {
    expect(parts(tx('inventory.empty', {}))).toEqual([
      'Rien ici pour l’instant. Ouvre un coffre pour trouver des pièces.',
    ]);
  });
});

describe('txn', () => {
  it('accorde le gabarit puis y insère le nœud', () => {
    const n = createElement('span', null, '3');
    expect(parts(txn('chest.pity', 3, { rank: 'Épique', n }))).toContain(n);
    const one = txn('chest.pity', 1, { rank: 'Épique', n: '1' });

```tsx
import { createElement, Fragment, type ReactElement, type ReactNode } from 'react';
import { raw, rawPlural, type MessageKey, type PluralKey } from './index';

/** Découpe le gabarit sur `{nom}` et interfole les nœuds. Les segments vides —
 *  aux bords, ou entre deux variables collées — sont écartés. */
function split(template: string, vars: Record<string, ReactNode>): ReactNode[] {
  return template
    .split(/\{(\w+)\}/g)
    // `split` avec un groupe capturant alterne littéral, nom, littéral, nom…
    .map((piece, i) => (i % 2 === 1 ? (piece in vars ? vars[piece] : `{${piece}}`) : piece))
    .filter((piece) => piece !== '');
}

/** Les parts sont passées en arguments variadiques et non en tableau : React
 *  traite alors les enfants comme statiques et n'exige pas de `key`. */
export function tx(key: MessageKey, vars: Record<string, ReactNode>): ReactElement {
  return createElement(Fragment, null, ...split(raw(key), vars));
}

export function txn(
  base: PluralKey, count: number, vars: Record<string, ReactNode> = {},
): ReactElement {
  return createElement(Fragment, null, ...split(rawPlural(base, count), { n: count, ...vars }));
}
```
